第 06 章

Agent 架构

前面几种架构里，「跑几步」都是你写死的。Agent 的分水岭在于：循环几次、什么时候停，由模型自己决定。这是目前最强、也最不可预测的一种架构。

读完这一章你会

  • 手工推演一遍 ReAct 的「计划—执行」循环，理解它到底在循环什么
  • 用 ToolNode + tools_condition 三十行写出一个能用的 Agent
  • 知道什么时候该「强制先调某个工具」，以及它的代价
  • 工具超过十几个时，用 RAG 先筛一遍工具

什么叫「Agent」

人工智能领域里 agent 的经典定义是「会行动的东西」。拆开来，「行动」隐含三件事：

  • 要能决定做什么
  • 决定意味着有多个选项——只有一个选项那不叫决定
  • 要能拿到外部环境的信息，否则没法判断

落到 LLM 上，就是两个提示技巧的组合：工具调用（告诉模型有哪些函数可用，以及怎么格式化调用请求）+ 思维链（让它一步步推理）。

计划—执行循环

用 LangGraph 写一个

LangGraph 提供了两个现成件，把上面的循环压缩成几行：

组件是做什么
ToolNode(tools)一个节点读最新那条 AIMessage 里的工具调用请求，逐个执行，把结果包成 ToolMessage 返回。工具抛异常也会被它捕获，转成错误内容的 ToolMessage 交给模型自己处理。
tools_condition一个条件边函数看最新 AIMessage 有没有工具调用：有 → 去 "tools"；没有 → END。这就是循环的出口。
ch6/{py,js}/a-basic-agent 
# pip install ddgs
import ast
from typing import Annotated, TypedDict

from langchain_community.tools import DuckDuckGoSearchRun
from langchain_core.messages import HumanMessage
from langchain_core.tools import tool
from langchain_openai import ChatOpenAI
from langgraph.graph import START, StateGraph
from langgraph.graph.message import add_messages
from langgraph.prebuilt import ToolNode, tools_condition


@tool
def calculator(query: str) -> str:
    """A simple calculator tool. Input should be a mathematical expression."""
    # ↑ 这句 docstring 就是模型看到的工具说明，务必写清楚
    return ast.literal_eval(query)


search = DuckDuckGoSearchRun()
tools = [search, calculator]

# bind_tools：把工具的名字/说明/参数 schema 一起发给模型
model = ChatOpenAI(model="gpt-4.1-mini", temperature=0.1).bind_tools(tools)


class State(TypedDict):
    messages: Annotated[list, add_messages]


def model_node(state: State) -> State:
    return {"messages": model.invoke(state["messages"])}


builder = StateGraph(State)
builder.add_node("model", model_node)
builder.add_node("tools", ToolNode(tools))
builder.add_edge(START, "model")
builder.add_conditional_edges("model", tools_condition)   # 出口
builder.add_edge("tools", "model")                        # 回边 = 循环

graph = builder.compile()

for c in graph.stream({"messages": [HumanMessage("美国第 30 任总统去世时多大年纪？")]}):
    print(c)
// npm i duck-duck-scrape expr-eval
import { DuckDuckGoSearch } from '@langchain/community/tools/duckduckgo_search';
import { Calculator } from '@langchain/community/tools/calculator';
import { ChatOpenAI } from '@langchain/openai';
import { HumanMessage } from '@langchain/core/messages';
import {
  StateGraph, Annotation, messagesStateReducer, START,
} from '@langchain/langgraph';
import { ToolNode, toolsCondition } from '@langchain/langgraph/prebuilt';

const tools = [new DuckDuckGoSearch(), new Calculator()];
const model = new ChatOpenAI({ model: 'gpt-4.1-mini', temperature: 0.1 }).bindTools(tools);

const annotation = Annotation.Root({
  messages: Annotation({ reducer: messagesStateReducer, default: () => [] }),
});

async function modelNode(state) {
  return { messages: await model.invoke(state.messages) };
}

const graph = new StateGraph(annotation)
  .addNode('model', modelNode)
  .addNode('tools', new ToolNode(tools))
  .addEdge(START, 'model')
  .addConditionalEdges('model', toolsCondition)
  .addEdge('tools', 'model')
  .compile();

变体一：强制先调某个工具

标准 Agent 每次都让 LLM 决定第一步。但如果你已经知道第一步必然是搜索，那第一次 LLM 调用就是浪费。

做法是加一个不调用 LLM 的节点，直接手工构造一条带工具调用的 AIMessage，然后直接进 tools：

ch6/{py,js}/b-force-first-tool 
from uuid import uuid4

from langchain_core.messages import AIMessage, ToolCall


def first_model(state: State) -> State:
    query = state["messages"][-1].content
    # 手工伪造一条「模型要求调用搜索」的消息，把用户原话直接当搜索词
    search_tool_call = ToolCall(name="duckduckgo_search", args={"query": query}, id=uuid4().hex)
    return {"messages": AIMessage(content="", tool_calls=[search_tool_call])}


builder = StateGraph(State)
builder.add_node("first_model", first_model)
builder.add_node("model", model_node)
builder.add_node("tools", ToolNode(tools))

builder.add_edge(START, "first_model")      # ← 入口变了
builder.add_edge("first_model", "tools")      # ← 直奔工具
builder.add_conditional_edges("model", tools_condition)
builder.add_edge("tools", "model")

graph = builder.compile()
import { AIMessage } from '@langchain/core/messages';

async function firstModelNode(state) {
  const query = state.messages.at(-1).content;
  const searchToolCall = {
    name: 'duckduckgo_search',
    args: { query },
    id: crypto.randomUUID(),
  };
  return { messages: [new AIMessage({ content: '', tool_calls: [searchToolCall] })] };
}

const graph = new StateGraph(annotation)
  .addNode('first_model', firstModelNode)
  .addNode('model', modelNode)
  .addNode('tools', new ToolNode(tools))
  .addEdge(START, 'first_model')
  .addEdge('first_model', 'tools')
  .addEdge('tools', 'model')
  .addConditionalEdges('model', toolsCondition)
  .compile();

什么时候不该这么做

只有当「第一步必然是这个工具」是一条确定的业务规则时才值得。如果只是「大多数时候是」，硬加这条规则反而会让那些不需要搜索的问题走一次无用的搜索，效果变差。这是典型的「用确定性换灵活性」。

变体二：工具太多

工具超过十来个，模型选错的概率明显上升——选项太多它会挑花眼。而且每次调用都要把所有工具的说明塞进提示，token 成本一直在涨。

解法很优雅：对工具本身做一次 RAG。把每个工具的描述当文档嵌入进向量库，用户提问时先检索出最相关的几个，只把这几个 bind_tools 给模型。

ch6/{py,js}/c-many-tools 
from langchain_core.documents import Document
from langchain_core.vectorstores.in_memory import InMemoryVectorStore
from langchain_openai import OpenAIEmbeddings

# 把每个工具的 description 当文档存进向量库，name 放 metadata
tools_retriever = InMemoryVectorStore.from_documents(
    [Document(t.description, metadata={"name": t.name}) for t in tools],
    OpenAIEmbeddings(model="text-embedding-3-small"),
).as_retriever()


class State(TypedDict):
    messages: Annotated[list, add_messages]
    selected_tools: list[str]              # ← 多一个字段存筛选结果


def select_tools(state: State) -> State:
    query = state["messages"][-1].content
    tool_docs = tools_retriever.invoke(query)
    return {"selected_tools": [doc.metadata["name"] for doc in tool_docs]}


def model_node(state: State) -> State:
    selected = [t for t in tools if t.name in state["selected_tools"]]
    # 注意：每次调用现场 bind_tools，绑的是这一轮筛出来的子集
    return {"messages": model.bind_tools(selected).invoke(state["messages"])}


builder = StateGraph(State)
builder.add_node("select_tools", select_tools)
builder.add_node("model", model_node)
builder.add_node("tools", ToolNode(tools))          # ToolNode 仍拿全量（执行时要认得）
builder.add_edge(START, "select_tools")
builder.add_edge("select_tools", "model")
builder.add_conditional_edges("model", tools_condition)
builder.add_edge("tools", "model")

graph = builder.compile()
import { Document } from '@langchain/core/documents';
import { MemoryVectorStore } from '@langchain/classic/vectorstores/memory';
import { OpenAIEmbeddings } from '@langchain/openai';

const toolsStore = await MemoryVectorStore.fromDocuments(
  tools.map((t) => new Document({ pageContent: t.description, metadata: { name: t.name } })),
  new OpenAIEmbeddings({ model: 'text-embedding-3-small' })
);
const toolsRetriever = toolsStore.asRetriever();

async function selectTools(state) {
  const query = state.messages.at(-1).content;
  const toolDocs = await toolsRetriever.invoke(query);
  return { selected_tools: toolDocs.map((d) => d.metadata.name) };
}

async function modelNode(state) {
  const selected = tools.filter((t) => state.selected_tools.includes(t.name));
  return { messages: await model.bindTools(selected).invoke(state.messages) };
}

小结

记住这个

Agent = 循环 + 模型控制的出口

去掉循环就是路由器，去掉条件边就死循环。两者缺一不可。

记住这个

工具的 docstring 是提示词

模型全靠它判断该不该调、怎么调。写得含糊，选错工具的锅在你。

记住这个

确定的规则就写死

「一定先搜索」这种业务确定性，硬编码比让 LLM 每次重新判断更快更稳。

记住这个

工具多了就先检索工具

超过 10 个开始考虑；代价是一次嵌入调用的延迟。
